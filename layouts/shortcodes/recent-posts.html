{{/* Recent Blog Posts Shortcode Displays recent blog posts in a card grid using Hextra styling. Parameters: - limit: Number of posts to show (3-6, default: 3) - cols: Number of grid columns (default: 3) - tag: Optional tag filter Example usage: recent-posts recent-posts limit=6 recent-posts cols=2 recent-posts limit=4 cols=2 recent-posts tag="tutorial" */}} {{- $limit := .Get "limit" | default 3 -}} {{- $cols := .Get "cols" | default 3 -}} {{- $tagFilter := .Get "tag" -}} {{/* Ensure limit is between 3 and 6 */}} {{- $limit = cond (lt $limit 3) 3 $limit -}} {{- $limit = cond (gt $limit 6) 6 $limit -}} {{/* Get blog posts - exclude drafts */}} {{- $blogPages := where site.RegularPages "Type" "blog" -}} {{- $blogPages = where $blogPages "Params.draft" "!=" true -}} {{/* Filter by tag if specified */}} {{- with $tagFilter -}} {{- $blogPages = where $blogPages "Params.tags" "intersect" (slice $tagFilter) -}} {{- end -}} {{/* Only proceed if we have pages */}} {{- if $blogPages -}} {{/* Sort by date descending and limit */}} {{- $blogPages = sort $blogPages "Date" "desc" -}} {{- $blogPages = first $limit $blogPages -}} {{- $imageStyle := "" -}} {{/* Render cards grid */}}
{{- range $blogPages }} {{- $post := . -}} {{- $featuredImage := "" -}} {{/* Try to get featured image from page resources */}} {{- with $post.Params.featuredImage -}} {{- $imgResource := $post.Resources.GetMatch . -}} {{- if $imgResource -}} {{- $featuredImage = $imgResource.RelPermalink -}} {{- end -}} {{- end -}} {{/* Also check for image in page resources if no frontmatter featuredImage */}} {{- if not $featuredImage -}} {{- $img := $post.Resources.GetMatch "{featured*,cover*,thumbnail*}" -}} {{- with $img -}} {{- $featuredImage = .RelPermalink -}} {{- end -}} {{- end -}} {{- $linkClass := "hx:hover:border-gray-300 hx:bg-transparent hx:shadow-xs hx:dark:border-neutral-800 hx:hover:bg-slate-50 hx:hover:shadow-md hx:dark:hover:border-neutral-700 hx:dark:hover:bg-neutral-900" -}} {{- with $featuredImage -}} {{- $linkClass = "hx:hover:border-gray-300 hx:bg-gray-100 hx:shadow-sm hx:dark:border-neutral-700 hx:dark:bg-neutral-800 hx:dark:text-gray-50 hx:hover:shadow-lg hx:dark:hover:border-neutral-500 hx:dark:hover:bg-neutral-700" -}} {{- end -}} {{- with $featuredImage -}} {{- end -}} {{- $post.Title -}}
